  <script type="module">
    async function initInfinityPanel() {
      const $w = $('.th-infinity-panel').not('.initialized').last();
      if ($w.length === 0) return;
      $w.addClass('initialized');

      await waitGlobalInitialized('Mvu');

      // 用于存储当前所有变量的引用，方便点击事件读取
      let currentVars = {};

      function renderAttributes(attrs, $container, isMini = false) {
        const attrKeys = ['力量', '敏捷', '耐力', '智力', '感知', '决心', '风度', '操控', '沉着'];
        let html = '';
        attrKeys.forEach(k => {
          const val = _.get(attrs, `${k}.当前`, 1);
          const leg = _.get(attrs, `${k}.传奇`, 0);
          const legHtml = leg > 0 ? `<span class="attr-leg">+${leg}</span>` : '';

          if (isMini) {
            html += `<span>${k}: <span style="color:var(--accent-cyan)">${val}</span>${legHtml}</span>`;
          } else {
            html += `<div class="attr-item glass-box">
              <span class="bar-label">${k}</span>
              <span class="attr-val">${val}${legHtml}</span>
            </div>`;
          }
        });
        $container.html(html);
      }

      function renderBar($bar, $text, current, max, pct) {
        $bar.css('width', `${Math.max(0, Math.min(100, pct))}%`);
        $text.text(`${current}/${max}`);
      }

      function updateUI() {
        const allVars = getAllVariables();
        currentVars = allVars;

        const char = _.get(allVars, 'stat_data.轮回者', null);
        if (!char) return;

        // 1. 头部
        $w.find('.js-location').text(_.get(allVars, 'stat_data.世界与任务.当前影片世界', '未知维度'));
        $w.find('.js-time').text(_.get(allVars, 'stat_data.世界与任务.当前时间', '--:--'));
        $w.find('.js-points').text(_.get(char, '核心资源.奖励点', 0));

        let branchText = '';
        const branches = _.get(char, '核心资源.支线剧情', null);
        if (branches) {
          ['S','A','B','C','D'].forEach(lvl => {
            if(branches[lvl]) branchText += `${lvl}:${branches[lvl]} `;
          });
        }
        $w.find('.js-branches').text(branchText || '无');

        // 2. 状态条
        const hp = _.get(char, '衍生状态.生命值', {完好:0, 上限:1});
        renderBar($w.find('.js-hp-bar'), $w.find('.js-hp-text'), hp.完好, hp.上限 || 1, (hp.完好 / (hp.上限 || 1)) * 100);

        const wp = _.get(char, '衍生状态.意志力', {当前:0, 上限:1});
        renderBar($w.find('.js-wp-bar'), $w.find('.js-wp-text'), wp.当前, wp.上限 || 1, (wp.当前 / (wp.上限 || 1)) * 100);

        // 3. 核心属性
        renderAttributes(_.get(char, '属性', {}), $w.find('.js-attr-grid'));

        // 4. 强化体系
        const powers = _.get(char, '能力体系.强化体系', {});
        let powerHtml = '';
        _.forEach(powers, (p, k) => {
          powerHtml += `<div class="list-item js-power-item" data-key="${_.escape(k)}">
            <span>${_.escape(p.名称 || k)} [${_.escape(p.等级 || '?')}]</span>
            <span style="color:var(--accent-cyan)">${_.escape(p.本质 || '')}</span>
          </div>`;
        });
        $w.find('.js-power-list').html(powerHtml || '<div style="text-align:center;color:var(--text-muted);font-size:0.8rem">暂无能力</div>');

        // 5. 物品装备 (包含负重)
        const load = _.get(char, '衍生状态.负重系统', {当前负重:0, 轻度上限:1, 负荷状态:'轻度'});
        let invHtml = `
          <div class="glass-box" style="margin-bottom:8px; font-size:0.8rem;">
            <div class="bar-label"><span>当前负重 (${load.负荷状态})</span><span>${load.当前负重}/${load.轻度上限}</span></div>
            <div class="progress-bg" style="margin-top:4px;"><div class="progress-fill fill-load" style="width:${Math.min(100, (load.当前负重 / (load.轻度上限 || 1)) * 100)}%"></div></div>
          </div>
        `;
        const inv = _.get(char, '物品与装备', {});
        const catMap = {
          '武器列表': '武器', '衣物列表': '衣物', '饰品列表': '饰品',
          '消耗品列表': '消耗品', '载具列表': '载具', '其他物品': '其他'
        };
        _.forEach(catMap, (label, listKey) => {
          const list = inv[listKey] || {};
          if (Object.keys(list).length > 0) {
            let itemsHtml = '';
            _.forEach(list, (item, k) => {
              itemsHtml += `<div class="list-item js-item-item" data-cat="${_.escape(listKey)}" data-key="${_.escape(k)}">
                <span>${_.escape(item.名称 || k)}</span>
                <span style="color:${item.当前装备 ? 'var(--accent-gold)' : 'var(--text-muted)'}">${item.当前装备 ? '已装备' : label}</span>
              </div>`;
            });
            invHtml += `
              <div class="accordion">
                <div class="accordion-header"><span>${label}</span> <span style="font-size:0.7rem">▼</span></div>
                <div class="accordion-content">${itemsHtml}</div>
              </div>
            `;
          }
        });
        $w.find('.js-inv-list').html(invHtml || '<div style="text-align:center;color:var(--text-muted);font-size:0.8rem">行囊空空</div>');

        // 6. 主神兑换
        const shop = _.get(allVars, 'stat_data.世界与任务.主神商店.当前展示列表', {});
        let shopHtml = '';
        _.forEach(shop, (desc, name) => {
          // 使用 _.escape 防止名字里有双引号等破坏 HTML
          shopHtml += `<div class="list-item js-shop-item" data-key="${_.escape(name)}">
            <span>${_.escape(name)}</span>
            <span style="color:var(--accent-gold)">查看</span>
          </div>`;
        });
        $w.find('.js-shop-list').html(shopHtml || '<div style="text-align:center;color:var(--text-muted);font-size:0.8rem">未连接主神或列表为空</div>');

        // 7. NPC
        const npcs = _.get(allVars, 'stat_data.人际关系列表', {});
        let npcHtml = '';
        _.forEach(npcs, (npc, k) => {
          npcHtml += `<div class="list-item js-npc-item" data-key="${_.escape(k)}">
            <span>${_.escape(npc.名称 || k)}</span>
            <span style="color:var(--accent-red)">好感: ${npc.好感度 || 0}</span>
          </div>`;
        });
        $w.find('.js-npc-list').html(npcHtml || '<div style="text-align:center;color:var(--text-muted);font-size:0.8rem">孤身一人</div>');
      }

      // --- 弹窗逻辑封装 ---
      function openModal(html) {
        $w.find('.js-modal-body').html(html);
        $w.find('.modal-overlay').css('display', 'flex');
      }

      // 绑定交互事件
      $w.find('.tab-btn').on('click', function() {
        $w.find('.tab-btn').removeClass('active');
        $(this).addClass('active');
        const target = $(this).data('target');
        $w.find('.tab-content').removeClass('active');
        $w.find(`.js-tab-${target}`).addClass('active');
      });

      $w.on('click', '.accordion-header', function() {
        $(this).next('.accordion-content').slideToggle(200);
      });

      // 弹窗关闭机制
      $w.find('.modal-overlay').on('click', function(e) {
        if (e.target === this) {
          $(this).hide();
        }
      });

      // 点击强化体系
      $w.on('click', '.js-power-item', function() {
        const key = $(this).attr('data-key');
        // 安全读取，避免 key 中的点号被 _.get 当作层级解析
        const powerList = _.get(currentVars, 'stat_data.轮回者.能力体系.强化体系', {});
        const p = powerList[key] || {};

        let html = `
          <h3 style="margin:0 0 12px 0;color:var(--accent-gold)">${_.escape(p.名称 || key)} [${_.escape(p.等级 || '')}]</h3>
          <div class="detail-row"><span class="detail-label">本质:</span> ${_.escape(p.本质 || '未知')}</div>
          <div class="detail-row"><span class="detail-label">类型:</span> ${_.escape(p.类型 || '未知')}</div>
          <div class="detail-row"><span class="detail-label">前提:</span> ${_.escape(p.前提 || '无')}</div>
          <div class="detail-row"><span class="detail-label">能量池:</span> ${_.escape(p.能量池 || '无')}</div>
          <div class="detail-row"><span class="detail-label">技能树:</span> ${_.escape(p.技能树 || '无')}</div>
          <div class="detail-row"><span class="detail-label">特殊:</span> ${_.escape(p.特殊 || '无')}</div>
          <div class="detail-desc">${_.escape(p.总描述 || '暂无描述')}</div>
        `;
        if (p.特性 && p.特性.length > 0) {
          html += `<div style="margin-top:12px;color:var(--accent-cyan)">[特性]</div><ul style="margin:4px 0;padding-left:20px;font-size:0.85rem;color:#ccc">`;
          p.特性.forEach(t => html += `<li>${_.escape(t)}</li>`);
          html += `</ul>`;
        }
        openModal(html);
      });

      // 点击物品
      $w.on('click', '.js-item-item', function() {
        const cat = $(this).attr('data-cat');
        const key = $(this).attr('data-key');

        const catList = _.get(currentVars, `stat_data.轮回者.物品与装备.${cat}`, {});
        const item = catList[key] || {};

        let html = `<h3 style="margin:0 0 12px 0;color:var(--accent-gold)">${_.escape(item.名称 || key)}</h3>`;
        if(item.体积 !== undefined) html += `<div class="detail-row"><span class="detail-label">体积:</span> ${item.体积}</div>`;
        if(item.重量 !== undefined) html += `<div class="detail-row"><span class="detail-label">重量:</span> ${item.重量}</div>`;
        if(item.武器伤害 !== undefined) html += `<div class="detail-row"><span class="detail-label">武器伤害:</span> ${item.武器伤害} ${item.伤害类型||''}</div>`;
        if(item.基本属性) html += `<div class="detail-row"><span class="detail-label">基本属性:</span> ${_.escape(item.基本属性)}</div>`;
        if(item.特殊属性) html += `<div class="detail-row"><span class="detail-label">特殊属性:</span> ${_.escape(item.特殊属性)}</div>`;

        html += `<div class="detail-desc">${_.escape(item.描述 || '暂无描述')}</div>`;

        if (item.特性 && item.特性.length > 0) {
          html += `<div style="margin-top:12px;color:var(--accent-cyan)">[特性]</div><ul style="margin:4px 0;padding-left:20px;font-size:0.85rem;color:#ccc">`;
          item.特性.forEach(t => html += `<li>${_.escape(t)}</li>`);
          html += `</ul>`;
        }

        // 装备切换选项
        if (['武器列表', '衣物列表', '饰品列表'].includes(cat)) {
          const checked = item.当前装备 ? 'checked' : '';
          html += `
            <div class="switch-container">
              <span style="font-size:0.85rem;color:var(--text-main)">装备状态</span>
              <label class="switch">
                <input type="checkbox" ${checked} disabled title="状态栏仅供查看，请在对话中向ST声明切换装备">
                <span class="slider"></span>
              </label>
            </div>
            <div style="font-size:0.7rem;color:var(--text-muted);text-align:right;margin-top:4px;">* 状态栏受限，请在聊天中声明切换</div>
          `;
        }
        openModal(html);
      });

      // 点击商店物品
      $w.on('click', '.js-shop-item', function() {
        const key = $(this).attr('data-key');
        // 安全读取，避免 key 中的点号被解析
        const shopList = _.get(currentVars, 'stat_data.世界与任务.主神商店.当前展示列表', {});
        const desc = shopList[key] || '';

        openModal(`
